# 1. Define the source files and your target physical directory
set(SOURCES 
    ${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp 
    ${CMAKE_CURRENT_SOURCE_DIR}/src/utils.cpp
)
set(TARGET_DIR ${CMAKE_CURRENT_BINARY_DIR}/Source_Files)

# 2. Create the physical directory on your disk
file(MAKE_DIRECTORY ${TARGET_DIR})

# 3. Copy the files into the new physical folder
foreach(FILE_PATH ${SOURCES})
    file(COPY ${FILE_PATH} DESTINATION ${TARGET_DIR})
endforeach()
